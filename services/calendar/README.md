# Calendar (Sky Alerts) service

RSS-based calendar of amateur astronomy events: meteors, eclipses, conjunctions, occultations, comets. **No weather** — astro-weather is a separate service.

- **Input**: RSS feeds (e.g. in-the-sky.org), optional seed events from `data/seed_events.yaml`.
- **Output**: JSON in `services/calendar/outputs/` (intermediate); the same JSONs are copied to `sites/staging/calendar/` (e.g. `/calendar/daily_signal.json`). Public RSS at `sites/staging/alerts/rss.xml`.

## Run

From repo root (recommended — uses project `.venv` and has PyYAML/feedparser):

```bash
make deps-calendar   # once (installs into .venv)
make calendar        # or make calendar-back
```

If you run the script directly (e.g. from IDE), use the **same** Python that has the deps: either activate the venv first (`source .venv/bin/activate`) or run via `make calendar`. Otherwise you may get `ModuleNotFoundError: No module named 'yaml'` when the IDE uses system `python3` without the venv packages.

## Config

- `configs/sources.yaml` — RSS streams (meteors, eclipses, conjunctions, occultations, comets; observations optional). No `astro_weather` stream.
- `configs/rules.yaml` — ranking, freshness, categorization. No `ui.weather` usage.

## Data flow

1. Pipeline fetches RSS per stream, normalizes to `NewsRecord`, filters, scores, merges.
2. Writes `daily_<stream>.json` and unified `daily_signal.json` to `outputs/`.
3. `render_rss` reads `daily_signal.json`, generates `sites/staging/alerts/rss.xml` (no weather items).
4. All `daily_*.json` from `outputs/` are copied to `sites/staging/calendar/` for the frontend.

Frontend: widget **alerts** uses `/alerts/rss.xml`; calendar JSONs are at `/calendar/daily_signal.json`, `/calendar/daily_meteors.json`, etc. (see `frontend/config/widgets.yaml`).

# -----------------------------
# Config

TZ = ZoneInfo("Europe/Warsaw")

DAYS = 7
STEP_MIN = 10

SITE_LAT_DEG = 52.2297
SITE_LON_DEG = 21.0122
SITE_ELEV_M  = 100.0

MAX_ITEMS = 30

# Scoring knobs
MIN_ALT_DEG = 10.0
MERIDIAN_BONUS_WINDOW = (time(2, 0), time(4, 0))  # local clock


@dataclass
class Candidate:
    id: str
    kind: str          # "calendar" | "planet" | "solar_system" | "dso"
    name: str
    ra_deg: float | None = None
    dec_deg: float | None = None
    # optional metadata
    mag: float | None = None
    meta: dict | None = None


def clamp(x: float, a: float, b: float) -> float:
    return max(a, min(b, x))


def load_json(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)


def night_times(observer: Observer, date_local: datetime) -> tuple[Time, Time]:
    """
    Returns civil dusk -> civil dawn for the given local date (Warsaw).
    """
    # Astroplan wants Time; use noon to avoid edge cases
    t0 = Time(datetime(date_local.year, date_local.month, date_local.day, 12, 0, tzinfo=TZ))
    dusk = observer.twilight_evening_civil(t0, which="nearest")
    dawn = observer.twilight_morning_civil(t0, which="next")
    return dusk, dawn


def make_time_grid(t_start: Time, t_end: Time, step_min: int) -> Time:
    n = int(np.ceil(((t_end - t_start).to(u.min).value) / step_min)) + 1
    dt = np.arange(n) * step_min * u.min
    return t_start + dt


def ha_deg_for_times(ra_deg: float, lst_deg: np.ndarray) -> np.ndarray:
    """
    Hour angle in degrees, wrapped to [-180..+180].
    HA = LST - RA
    """
    ha = (lst_deg - ra_deg + 540.0) % 360.0 - 180.0
    return ha


def az_sector(az_deg: float) -> str:
    # crude but good enough for narrative
    if az_deg < 45 or az_deg >= 315:
        return "north"
    if az_deg < 135:
        return "east"
    if az_deg < 225:
        return "south"
    return "west"


def score_visibility(
    alt_deg: np.ndarray,
    ha_deg: np.ndarray,
    moon_sep_deg: np.ndarray | None,
    times_local: np.ndarray,
) -> tuple[float, dict]:
    """
    alt_deg: altitude series for the night
    ha_deg:  hour angle series for the night
    moon_sep_deg: optional series, same length
    times_local: array of datetimes in local TZ, same length
    """
    above0 = alt_deg > 0
    above10 = alt_deg >= 10
    above20 = alt_deg >= 20

    minutes_above_10 = int(np.sum(above10) * STEP_MIN)
    alt_max = float(np.max(alt_deg)) if np.any(above0) else -90.0
    alt_mean = float(np.mean(alt_deg[above0])) if np.any(above0) else -90.0

    # Meridian bonus evaluated in 02:00–04:00 local
    t_lo, t_hi = MERIDIAN_BONUS_WINDOW
    in_mer_win = np.array([(t_lo <= dt.time() <= t_hi) for dt in times_local], dtype=bool) & above0
    if np.any(in_mer_win):
        min_abs_ha = float(np.min(np.abs(ha_deg[in_mer_win])))
    else:
        min_abs_ha = 180.0

    # Moon separation term
    if moon_sep_deg is not None and len(moon_sep_deg) == len(alt_deg) and np.any(above0):
        min_moon_sep = float(np.min(moon_sep_deg[above0]))
    else:
        min_moon_sep = 90.0  # neutral

    S_alt = clamp(alt_max / 60.0, 0.0, 1.0) * 40.0
    S_time = clamp(minutes_above_10 / 180.0, 0.0, 1.0) * 25.0
    S_mer = clamp(1.0 - (min_abs_ha / 30.0), 0.0, 1.0) * 15.0
    S_moon = clamp(min_moon_sep / 90.0, 0.0, 1.0) * 10.0

    total = S_alt + S_time + S_mer + S_moon

    breakdown = {
        "alt_max_deg": alt_max,
        "alt_mean_deg": alt_mean,
        "minutes_above_10": minutes_above_10,
        "minutes_above_20": int(np.sum(above20) * STEP_MIN),
        "min_abs_ha_deg_02_04": min_abs_ha,
        "min_moon_sep_deg": min_moon_sep,
        "S_alt": S_alt,
        "S_time": S_time,
        "S_meridian": S_mer,
        "S_moon": S_moon,
        "S_total_geom": total,
    }
    return total, breakdown


def category_bonus(kind: str) -> float:
    return {
        "calendar": 30.0,
        "planet": 25.0,
        "solar_system": 10.0,
        "dso": 5.0,
    }.get(kind, 0.0)


def build_visibility_note_en(name: str, best_start: datetime, best_end: datetime, az_deg: float, alt_max: float) -> str:
    sector = az_sector(az_deg)
    t1 = best_start.strftime("%H:%M")
    t2 = best_end.strftime("%H:%M")
    alt_txt = f"peaking at ~{alt_max:.0f}°"
    return f"Best between {t1}–{t2} local time, {alt_txt}, {sector}ern sky."


def pick_best_window(times_local: np.ndarray, alt_deg: np.ndarray, az_deg: np.ndarray) -> tuple[datetime, datetime, float]:
    """
    Finds a simple best window: longest consecutive run above MIN_ALT_DEG,
    prefers later-night if tie (good for DSO).
    Returns: start, end, representative az_deg (median in window)
    """
    ok = alt_deg >= MIN_ALT_DEG
    if not np.any(ok):
        # fallback to max altitude sample
        i = int(np.argmax(alt_deg))
        dt = times_local[i]
        return dt, dt + timedelta(minutes=STEP_MIN), float(az_deg[i])

    idx = np.where(ok)[0]
    # group consecutive indices
    groups = []
    start = idx[0]
    prev = idx[0]
    for i in idx[1:]:
        if i == prev + 1:
            prev = i
        else:
            groups.append((start, prev))
            start = i
            prev = i
    groups.append((start, prev))

    # pick best: longest, tie -> later start
    best = max(groups, key=lambda ab: ((ab[1] - ab[0]), ab[0]))
    a, b = best
    return times_local[a], times_local[b] + timedelta(minutes=STEP_MIN), float(np.median(az_deg[a:b+1]))


def main():
    loc = EarthLocation(lat=SITE_LAT_DEG*u.deg, lon=SITE_LON_DEG*u.deg, height=SITE_ELEV_M*u.m)
    observer = Observer(location=loc, timezone=TZ, name="site")

    # ---------------------------------------
    # 1) Load calendar
    # ---------------------------------------
    cal = load_json(CALENDAR_PATH)
    cal_items = cal.get("items", [])  # daily_signal.json has meta + items  [oai_citation:1‡daily_signal.json](sediment://file_00000000730471f4b7953366b7e6fa97)

    # Keep only items relevant to next DAYS
    now_local = datetime.now(TZ)
    end_local = now_local + timedelta(days=DAYS)

    def parse_published(d: str) -> datetime | None:
        try:
            # daily_signal uses YYYY-MM-DD
            return datetime.strptime(d, "%Y-%m-%d").replace(tzinfo=TZ)
        except Exception:
            return None

    upcoming_calendar = []
    for it in cal_items:
        d = parse_published(it.get("published_at", ""))
        if not d:
            continue
        # Keep a window: from 1 day ago to +DAYS (so “today / recent” still can show)
        if (now_local - timedelta(days=1)) <= d <= end_local:
            upcoming_calendar.append(it)

    # Convert calendar items to candidates (no RA/Dec necessarily)
    calendar_candidates: list[Candidate] = []
    for it in upcoming_calendar:
        calendar_candidates.append(Candidate(
            id=it.get("id", ""),
            kind="calendar",
            name=it.get("title", "Astronomy event"),
            ra_deg=None,
            dec_deg=None,
            meta={"calendar": it},
        ))

    # ---------------------------------------
    # 2) Load planets (already computed ephemerides)
    # ---------------------------------------
    planets_json = load_json(PLANETS_PATH)
    # We'll turn each planet into a candidate with RA/Dec taken from the “nearest frame”
    # (exact picking logic can reuse your existing parse/pick helpers; here we take first frame as placeholder)
    # TODO: replace with your actual nearest-frame picker from generator logic
    frames = planets_json.get("frames", [])
    frame0 = frames[0] if frames else {}
    planets_map = (frame0.get("planets") or {})

    planet_candidates: list[Candidate] = []
    for key, p in planets_map.items():
        try:
            planet_candidates.append(Candidate(
                id=f"planet:{key}",
                kind="planet",
                name=p.get("name", key.title()),
                ra_deg=float(p["ra_deg"]),
                dec_deg=float(p["dec_deg"]),
                mag=float(p["mag"]) if p.get("mag") is not None else None,
                meta={"planet_key": key},
            ))
        except Exception:
            continue

    # ---------------------------------------
    # 3) DSO shortlist (placeholder)
    # ---------------------------------------
    # TODO: replace with your curated list file: messier + a few seasonal NGC
    dso_candidates: list[Candidate] = [
        Candidate(id="m42", kind="dso", name="M42 (Orion Nebula)", ra_deg=83.8221/1.0, dec_deg=-5.3911/1.0),
        Candidate(id="m31", kind="dso", name="M31 (Andromeda Galaxy)", ra_deg=10.6847, dec_deg=41.2690),
        Candidate(id="m87", kind="dso", name="M87 (Virgo A)", ra_deg=187.7059, dec_deg=12.3911),
    ]

    # ---------------------------------------
    # 4) Scoring over tonight’s window (civil dusk -> civil dawn)
    # ---------------------------------------
    dusk, dawn = night_times(observer, now_local)
    grid = make_time_grid(dusk, dawn, STEP_MIN)

    # Convert grid to local datetimes for meridian window logic
    grid_dt_utc = grid.to_datetime(timezone=ZoneInfo("UTC"))
    grid_dt_local = np.array([dt.astimezone(TZ) for dt in grid_dt_utc], dtype=object)

    # LST degrees for HA calculation (astropy can give sidereal time)
    # Using astropy: local sidereal time at location
    lst = Time(grid_dt_utc).sidereal_time("apparent", longitude=SITE_LON_DEG*u.deg).degree
    lst_deg = np.array(lst, dtype=float)

    # Moon position for separation penalty (astropy built-in ephemeris is ok to start)
    # If you prefer, you can use your sun_moon.json instead.
    from astropy.coordinates import get_body
    moon_icrs = get_body("moon", Time(grid_dt_utc))
    moon_altaz = moon_icrs.transform_to(AltAz(obstime=Time(grid_dt_utc), location=loc))

    def eval_candidate(c: Candidate) -> dict | None:
        # Calendar items: no RA/Dec => we still “pin” them, but they won’t have geometry.
        if c.ra_deg is None or c.dec_deg is None:
            return {
                "id": c.id,
                "type": "event",
                "name": c.name,
                "origin": c.kind,
                "score": 1000.0,  # pin-grade; later we’ll sort calendar separately anyway
                "note_en": (c.meta or {}).get("calendar", {}).get("summary", "") or "Upcoming sky event.",
                "meta": c.meta or {},
            }

        sc = SkyCoord(ra=c.ra_deg*u.deg, dec=c.dec_deg*u.deg, frame="icrs")
        altaz = sc.transform_to(AltAz(obstime=Time(grid_dt_utc), location=loc))
        alt_deg = np.array(altaz.alt.degree, dtype=float)
        az_deg  = np.array(altaz.az.degree, dtype=float)

        # Moon separation
        sep = sc.separation(moon_icrs).degree
        sep = np.array(sep, dtype=float)

        ha = ha_deg_for_times(c.ra_deg, lst_deg)

        geom_score, breakdown = score_visibility(alt_deg, ha, sep, grid_dt_local)

        total = geom_score + category_bonus(c.kind)

        # pick best window + note
        w_start, w_end, w_az = pick_best_window(grid_dt_local, alt_deg, az_deg)
        note = build_visibility_note_en(c.name, w_start, w_end, w_az, breakdown["alt_max_deg"])

        return {
            "id": c.id,
            "type": "planet" if c.kind == "planet" else "dso",
            "name": c.name,
            "origin": c.kind,
            "ra_deg": c.ra_deg,
            "dec_deg": c.dec_deg,
            "score": round(float(total), 2),
            "score_geom": round(float(geom_score), 2),
            "score_breakdown": breakdown,
            "best_window_local": {"start": w_start.isoformat(), "end": w_end.isoformat()},
            "note_en": note,
            "meta": c.meta or {},
        }

    candidates = calendar_candidates + planet_candidates + dso_candidates

    evaluated = [eval_candidate(c) for c in candidates]
    evaluated = [e for e in evaluated if e is not None]

    # Split calendar “pins” and normal objects
    calendar_out = [e for e in evaluated if e.get("origin") == "calendar"]
    normal_out   = [e for e in evaluated if e.get("origin") != "calendar"]

    # Sort normal by score desc and cap
    normal_out.sort(key=lambda x: x.get("score", 0.0), reverse=True)
    normal_out = normal_out[:MAX_ITEMS]

    # Calendar goes on top (already filtered to next DAYS)
    out_items = calendar_out + normal_out

    out = {
        "version": 1,
        "generated_at": datetime.now(ZoneInfo("UTC")).isoformat(),
        "site": {"lat_deg": SITE_LAT_DEG, "lon_deg": SITE_LON_DEG, "tz": str(TZ)},
        "window": {"civil_dusk_utc": dusk.utc.iso, "civil_dawn_utc": dawn.utc.iso, "step_min": STEP_MIN},
        "items": out_items,
    }

    OUT_PATH_SERVICES.parent.mkdir(parents=True, exist_ok=True)
    OUT_PATH_STAGING.parent.mkdir(parents=True, exist_ok=True)

    OUT_PATH_SERVICES.write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")
    OUT_PATH_STAGING.write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")

    print(f"[ok] wrote {len(out_items)} items ->")
    print(f" - {OUT_PATH_SERVICES}")
    print(f" - {OUT_PATH_STAGING}")


if __name__ == "__main__":
    main()
